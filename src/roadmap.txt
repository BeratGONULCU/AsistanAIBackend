Kodu dikkatli kullanın.2. Adım: .NET SemanticResolver Servisini KodlamaBu servis, yapay zekaya gitmeden önce cümleyi çok hızlı bir şekilde yerel kurallarla çözmeye çalışır. Böylece hem bütçeden tasarruf edersin hem de sistem milisaniyeler içinde cevap verir.csharppublic class SemanticResolver
{
    private readonly MyDbContext _context;
    // Soru sinyallerini içeren lokal liste (İleride DB'ye de taşınabilir)
    private readonly List<string> _questionSignals = new() { "nedir", "ne demek", "kaç", "nasıl", "söyle", "neden" };
    private readonly List<string> _commandSignals = new() { "aç", "kapat", "çalıştır", "sil", "başlat" };

    public SemanticResolver(MyDbContext context) => _context = context;

    public async Task<SemanticOutput> ResolveLocalAsync(string rawText)
    {
        var cleanText = rawText.ToLower().Trim();
        var output = new SemanticOutput();

        // 1. Soru mu? Komut mu? Tespiti
        if (_questionSignals.Any(signal => cleanText.Contains(signal)))
        {
            output.Type = "question";
            output.Operation = "get_info";
        }
        else if (_commandSignals.Any(signal => cleanText.Contains(signal)))
        {
            output.Type = "command";
            // Operasyonu cümlenin sonundaki kelimeye göre kabaca eşleştirebiliriz
            if (cleanText.Contains("aç") || cleanText.Contains("başlat")) output.Operation = "open";
            if (cleanText.Contains("kapat") || cleanText.Contains("sil")) output.Operation = "close";
        }

        // 2. Target ve Alias (Takma Ad) Tespiti
        // DB'deki tüm alias'ları çekip cümle içinde geçiyor mu diye bakıyoruz
        var matchedAlias = await _context.TargetAliases
            .Include(a => a.Target)
            .FirstOrDefaultAsync(a => cleanText.Contains(a.AliasText.ToLower()));

        if (matchedAlias != null)
        {
            output.Target = matchedAlias.Target.SystemName;
            output.Domain = matchedAlias.Target.DefaultDomain;
            output.Confidence = 90; // Yerel kuralla bulduğumuz için güven yüksek
        }

        // Eğer hem tipi hem de hedefi yerel olarak bulabildiysek nesneyi dönüyoruz
        if (!string.IsNullOrEmpty(output.Type) && !string.IsNullOrEmpty(output.Target))
        {
            return output;
        }

        // Yerel kurallar cümleyi çözmeye yetmediyse null dön ki AI devreye girsin
        return null; 
    }
}

Bu sayede model asla serbest metin yazamaz, tam senin istediğin nesneyi üretir.csharp// OpenAI SDK kullanarak yapay zekadan nesne isteme örneği
public async Task<SemanticOutput> ResolveWithAIAsync(string rawText)
{
    var client = new OpenAIClient("YOUR_OPENAI_API_KEY");
    var chatClient = client.GetChatClient("gpt-4o-mini"); // Hem ucuz hem çok hızlı

    string systemPrompt = @"Kullanıcının doğal dilde verdiği cümleyi analiz et ve strictly belirtilen JSON formatına dönüştür.
    Kurallar:
    - type: 'question', 'command', 'chat', 'unknown'
    - domain: 'system', 'application', 'web', 'file', 'general'
    - target: Cümlenin hedef aldığı nesne (Örn: terminal, chrome, ram, camera)
    - operation: Yapılacak eylem (Örn: get_info, open, close, search)";

    // ChatCompletionOptions ile çıktı formatını kesin olarak JSON şemamıza zorluyoruz
    ChatCompletionOptions options = new()
    {
        ResponseFormat = ChatResponseFormat.CreateJsonSchemaFormat(
            jsonSchemaName: "semantic_resolution",
            jsonSchema: BinaryData.FromString(typeof(SemanticOutput).GetJsonSchema()) // Sınıfının şeması
        )
    };

    ChatCompletion completion = await chatClient.CompleteChatAsync(
        new[] {
            new SystemChatMessage(systemPrompt),
            new UserChatMessage(rawText)
        }, 
        options
    );

    // Gelen veri doğrudan şemaya uygun temiz bir JSON'dır
    string jsonResult = completion.Content[0].Text;
    return JsonSerializer.Deserialize<SemanticOutput>(jsonResult);
}

public async Task<IActionResult> ResolveCommand([FromBody] CommandRequest request)
{
    // 1. Önce yerel kurallarla hızlıca çözmeyi dene
    var result = await _semanticResolver.ResolveLocalAsync(request.RawText);
    
    // 2. Yerel kural bulamadıysa Yapay Zekaya (OpenAI/Gemini) sor
    if (result == null)
    {
        result = await _aiResolver.ResolveWithAIAsync(request.RawText);
    }

    // 3. Çıkan sonucu DB'deki SemanticActions tablosunda eşleştir
    var action = await _context.SemanticActions
        .FirstOrDefaultAsync(a => a.Type == result.Type && 
                                  a.Domain == result.Domain && 
                                  a.Target == result.Target && 
                                  a.Operation == result.Operation);

    if (action == null)
    {
        return Ok(new { success = false, message = "Bu işlemi nasıl yapacağımı henüz bilmiyorum." });
    }

    // 4. Onay mekanizması ve çalıştırma adımlarına geç (Faz 6'daki mantık)
    return Ok(ExecuteSemanticAction(action, result.Params));
}
